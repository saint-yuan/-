        /* Tab Switch and Search Controller */
        let currentTab = 'all';

        function renderNews() {
            const newsGrid = document.getElementById('newsGrid');
            const searchKeyword = document.getElementById('globalSearch').value.toLowerCase().trim();
            
            newsGrid.innerHTML = '';

            const filteredData = newsData.filter(item => {
                const matchesTab = (currentTab === 'all') || (item.type === currentTab);
                const matchesSearch = item.title.toLowerCase().includes(searchKeyword) ||
                                      item.content.toLowerCase().includes(searchKeyword) ||
                                      item.catText.toLowerCase().includes(searchKeyword);
                return matchesTab && matchesSearch;
            });

            if (filteredData.length === 0) {
                newsGrid.innerHTML = '<div class="no-results"><i class="fa-solid fa-scroll" style="font-size:2rem; margin-bottom:10px; display:block;"></i>查無相關公告訊息</div>';
                return;
            }

            filteredData.forEach(item => {
                const card = document.createElement('div');
                card.className = 'news-card';
                card.onclick = () => openModal(item.catText, item.title, `${item.content}<br><br><small style="color:var(--ink-grey)">發布日期：${item.date}</small>`);

                card.innerHTML = `
                    <div>
                        <div class="news-meta">
                            <span class="news-category">❖ ${item.catText}</span>
                            <span class="news-date"><i class="fa-regular fa-calendar"></i> ${item.date}</span>
                        </div>
                        <h3 class="news-title">${item.title}</h3>
                    </div>
                    <div class="news-footer">
                        閱讀詳細內容 <i class="fa-solid fa-arrow-right"></i>
                    </div>
                `;

                newsGrid.appendChild(card);
            });
        }

        function switchTab(tabType) {
            currentTab = tabType;
            const buttons = document.querySelectorAll('.tab-btn');
            buttons.forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            renderNews();
        }

        let currentSlide = 0;

        slides.forEach((_, idx) => {
            const dot = document.createElement('div');
            dot.className = `dot ${idx === 0 ? 'active' : ''}`;
            dot.onclick = () => goToSlide(idx);
            dotsContainer.appendChild(dot);

        const dots = document.querySelectorAll('.dot');

        function updateSlide() {
            slides.forEach((slide, idx) => {
                slide.classList.remove('active');
                dots[idx].classList.remove('active');
            });
            slides[currentSlide].classList.add('active');
            dots[currentSlide].classList.add('active');
        }

        function moveSlide(direction) {
            currentSlide = (currentSlide + direction + slides.length) % slides.length;
            updateSlide();
        }

        function goToSlide(index) {
            currentSlide = index;
            updateSlide();
        }
