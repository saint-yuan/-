        function toggleLanguage() {
            currentLang = currentLang === 'zh' ? 'en' : 'zh';
            document.getElementById('langText').innerText = currentLang === 'zh' ? 'EN' : '中文';
            
            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (translations[currentLang][key]) {
                    el.innerText = translations[currentLang][key];
                }
            });

            document.querySelectorAll('[data-i18n-ph]').forEach(el => {
                const key = el.getAttribute('data-i18n-ph');
                if (translations[currentLang][key]) {
                    el.placeholder = translations[currentLang][key];
                }
            });
        }

        // Works Render Logic
        let activeCategory = 'ALL';

        function renderWorks(items) {
            const grid = document.getElementById('worksGrid');
            grid.innerHTML = '';

            if (items.length === 0) {
                grid.innerHTML = `
                    <div style="grid-column: 1/-1; text-align: center; padding: 4rem; color: var(--ink-slate);">
                        <i class="fa-regular fa-folder-open" style="font-size: 3rem; margin-bottom: 1rem; color: var(--antique-gold); display: block;"></i>
                        未搜尋到符合條件之學生作品
                    </div>`;
                return;
            }

            items.forEach(item => {
                const card = document.createElement('div');
                card.className = 'work-card';

                card.innerHTML = `
                    <div class="work-cover" style="background-image: url('${item.image}');">
                        <div class="work-cover-overlay">
                            <span class="seal-badge ${item.sealClass}">${item.categoryName}</span>
                        </div>
                    </div>
                    <div class="work-body">
                        <h3 class="work-title">${item.title}</h3>
                        <div class="work-authors"><i class="fa-solid fa-users"></i> ${item.authors} | 指導：${item.advisor}</div>
                        <p class="work-desc">${item.desc}</p>
                        <div class="work-footer">
                            <button class="btn-classical" onclick="openProjectModal('${item.id}')">
                                詳閱報告 <i class="fa-solid fa-angle-right"></i>
                            </button>
                            <button class="stamp-btn ${item.stamped ? 'stamped' : ''}" id="stamp-btn-${item.id}" onclick="toggleStamp('${item.id}', event)">
                                <i class="fa-solid fa-stamp"></i> 賞讚 <span id="like-count-${item.id}">${item.likes}</span>
                            </button>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function filterWorks(category) {
            activeCategory = category;
            const buttons = document.querySelectorAll('.filter-tabs .tab-btn');
            buttons.forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');

            document.getElementById('workSearchInput').value = '';

            if (category === 'ALL') {
                renderWorks(studentWorksData);
            } else {
                const filtered = studentWorksData.filter(i => i.category === category);
                renderWorks(filtered);
            }
        }

        function searchWorks() {
            const query = document.getElementById('workSearchInput').value.toLowerCase();
            const filtered = studentWorksData.filter(item => {
                const matchesCat = activeCategory === 'ALL' || item.category === activeCategory;
                const matchesQuery = item.title.toLowerCase().includes(query) ||
                                     item.authors.toLowerCase().includes(query) ||
                                     item.advisor.toLowerCase().includes(query) ||
                                     item.desc.toLowerCase().includes(query);
                return matchesCat && matchesQuery;
            });
            renderWorks(filtered);
        }

        // Interactive Red Stamp Like Counter Function
        function toggleStamp(id, event) {
            event.stopPropagation();
            const item = studentWorksData.find(w => w.id === id);
            if (!item) return;

            const btn = document.getElementById(`stamp-btn-${id}`);
            const countEl = document.getElementById(`like-count-${id}`);

            if (!item.stamped) {
                item.stamped = true;
                item.likes += 1;
                btn.classList.add('stamped');
            } else {
                item.stamped = false;
                item.likes -= 1;
                btn.classList.remove('stamped');
            }
            countEl.innerText = item.likes;
        }

        // Modal Popups
        function openProjectModal(id) {
            const item = studentWorksData.find(w => w.id === id);
            if (!item) return;

            document.getElementById('modalCategory').innerText = item.categoryName;
            document.getElementById('modalCategory').className = `seal-badge ${item.sealClass}`;
            document.getElementById('modalTitle').innerText = item.title;
            document.getElementById('modalAuthors').innerText = item.authors;
            document.getElementById('modalAdvisor').innerText = item.advisor;
            document.getElementById('modalYear').innerText = item.year;
            document.getElementById('modalAward').innerText = item.award;
            document.getElementById('modalBody').innerHTML = item.content;

            document.getElementById('projectModal').classList.add('active');
        }

        function closeModal() {
            document.getElementById('projectModal').classList.remove('active');
        }

        window.onclick = function(e) {
            const modal = document.getElementById('projectModal');
            if (e.target === modal) closeModal();
        };

            document.getElementById('navLinks').classList.toggle('active');
